## Art.ts 改动
- preload()：resources.loadDir('art', SpriteFrame) 全部加载；所有 SpriteFrame 的 texture 设置 setFilters(Texture2D.Filter.NEAREST, Texture2D.Filter.NEAREST) 保证像素清晰；有 insets 的设置 spriteFrame.insetLeft/Right/Top/Bottom。然后后台 loadBundle('bg') 并加载所有背景。
- frames(name)：先查 ART_ALIAS 解析到真实路径，再取帧（res 的 art/<path>/0..n 按数字排序；bg 单帧）；icon_<itemId> 走 ICON_MAP → art/icons/<iconName>。
- makeVisual(name,w,h,color,label)：有帧→Sprite（sizeMode CUSTOM，宽高为 w,h；有 insets 时 type=SLICED）并按 fps(默认10) 播放，loopFrom 指定循环起点；无帧→保持原来的色块+文字占位。如果是 bg 且还没加载完，先占位，加载完自动替换成 Sprite。
- 新增 static makeCard(kind:'card_day'|'card_fail'|'card_cert', parentH:number):Node：卡片按高度适配：高度 = parentH*0.7，宽 = 高*0.75，水平居中两边留空；底图为卡片素材，上面挂一个 content 子节点供写文字。

- 新增 static drawCell(g:Graphics, x, y, size, state:'normal'|'hover'|'bad'|'filled')：代码画单个背包格（深色底+2px 浅色边框，hover 黄边，bad 红边）。
- 新增 static missing():string[] 返回本次运行中被请求但没有素材、走了占位的逻辑名（去重），用于报告。
保持原有公开方法签名兼容（preload/frames/makeVisual/play）。

export class Rng {
  private state: number;

  constructor(seed: number) {
    this.state = seed >>> 0;
  }

  next(): number {
    let t = (this.state += 0x6d2b79f5);
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  }

  int(a: number, b: number): number {
    const min = Math.ceil(a);
    const max = Math.floor(b);
    if (min > max) return min;
    return Math.floor(this.next() * (max - min + 1)) + min;
  }

  pick<T>(arr: T[]): T {
    if (arr.length === 0) throw new Error('pick from empty array');
    return arr[Math.floor(this.next() * arr.length)];
  }

  chance(p: number): boolean {
    return this.next() < p;
  }
}

      const boxes: Box[] = [];
      for (let i = 0; i < count; i++) {
        const size: BoxSize = this.rng.chance(this.cfg.bigBoxChance[tier]) ? 'big' : 'small';
        boxes.push({ id: `box_${loc.id}_${i}`, size, opened: false });
      }
      this.boxes.set(loc.id, boxes);
    }
    this.phase = 'map';
    this.bannedToday = [];
  }

  boxesAt(locId: string): Box[] {
    return this.boxes.get(locId) ?? [];
  }

  travelCost(locId: string): number {
    const loc = this.findLocation(locId);
    if (!loc) return this.cfg.maxStamina * 10;
    let cost: number;
    if (this.location == null) {
      cost = loc.distance;
    } else {
      const curLoc = this.findLocation(this.location);
      if (!curLoc) cost = loc.distance;
      else cost = Math.abs(curLoc.distance - loc.distance) + 5;
    }
    cost = Math.max(5, cost);

    const phaseDef = this.currentPhaseDef();
    if (loc.far && phaseDef.farTravelMul !== 1) {
      cost = Math.floor(cost * phaseDef.farTravelMul);
    }
    if (this.fortune?.effect === 'windy') {
      cost += this.cfg.fortuneEffects.windy;
    }
    if (this.raining) {
      cost += this.cfg.rainTravelExtra;
    }
    return Math.max(5, cost);
  }

  homeCost(): number {
    if (this.location == null) return 0;
    const loc = this.findLocation(this.location);
    if (!loc) return 0;
    let cost = loc.distance;
    const phaseDef = this.currentPhaseDef();
    if (loc.far && phaseDef.farTravelMul !== 1) {
      cost = Math.floor(cost * phaseDef.farTravelMul);
    }
    if (this.fortune?.effect === 'windy') {
      cost += this.cfg.fortuneEffects.windy;
    }
    if (this.raining) {
      cost += this.cfg.rainTravelExtra;
    }
    return Math.max(5, cost);
  }

  travel(locId: string): boolean {
    if (this.phase !== 'map' && this.phase !== 'location') return false;
    if (this.bannedToday.includes(locId)) return false;
    const loc = this.findLocation(locId);
    if (!loc) return false;

    const cost = this.travelCost(locId);
    this.stamina -= cost;
    this.minute += this.cfg.travelMinutes;
    this.location = locId;
    this.phase = 'location';

    if (this.stamina <= 0) {
      this.collapse();
      return true;
    }

    this.checkMidnight();
    return true;
  }

  goldChance(holdSec: number, boxId: string): number {
    const box = this.findBox(boxId);
    if (!box || !this.location) return 0;
    const clampHold = Math.min(Math.max(0, holdSec), this.cfg.maxHoldSec);
    let goodP = 0;
    let rareP = 0;
    let epicP = 0;
    for (const step of this.cfg.holdCurve) {
      if (step.minSec <= clampHold) {
        goodP = step.good;
        rareP = step.rare;
        epicP = step.epic;
      } else break;
    }

    const locId = this.location;
    const loc = this.findLocation(locId);
    if (!loc) return 0;
    const phaseDef = this.currentPhaseDef();
    let mul = loc.lootRate * phaseDef.lootMul;
    if (this.fortune?.effect === 'lootUp' && this.fortuneLocation === locId) {
      mul *= 1 + this.cfg.fortuneEffects.lootUp;
    }
    if (this.catLuck > 0) {
      mul *= 1 + this.cfg.catLuckMul;
    }
    if (this.mood < this.cfg.moodGoldLow.threshold) {
      mul *= this.cfg.moodGoldLow.mul;
    } else if (this.mood >= this.cfg.moodGoldHigh.threshold) {
      mul *= this.cfg.moodGoldHigh.mul;
    }

    rareP *= mul;
    epicP *= mul;
    rareP = Math.min(1, Math.max(0, rareP));
    epicP = Math.min(1, Math.max(0, epicP));
    return Math.min(1, rareP + epicP);
  }

  glowFor(holdSec: number, boxId: string): 'white' | 'blue' | 'gold' | 'rainbow' {
    const box = this.findBox(boxId);
    if (!box || !this.location) return 'white';
    const clampHold = Math.min(Math.max(0, holdSec), this.cfg.maxHoldSec);
    let goodP = 0;
    let rareP = 0;
    let epicP = 0;
    for (const step of this.cfg.holdCurve) {
      if (step.minSec <= clampHold) {
        goodP = step.good;
        rareP = step.rare;
        epicP = step.epic;
      } else break;
    }

    const locId = this.location;
    const loc = this.findLocation(locId);
    if (!loc) return 'white';
    const phaseDef = this.currentPhaseDef();
    let mul = loc.lootRate * phaseDef.lootMul;
    if (this.fortune?.effect === 'lootUp' && this.fortuneLocation === locId) {
      mul *= 1 + this.cfg.fortuneEffects.lootUp;
    }
    if (this.catLuck > 0) {
      mul *= 1 + this.cfg.catLuckMul;
    }
    if (this.mood < this.cfg.moodGoldLow.threshold) {
      mul *= this.cfg.moodGoldLow.mul;
    } else if (this.mood >= this.cfg.moodGoldHigh.threshold) {
      mul *= this.cfg.moodGoldHigh.mul;
    }

    goodP *= mul;
    rareP *= mul;
    epicP *= mul;
    if (box.size === 'big') {
      goodP += this.cfg.bigBoxBonusGood;
    }
    goodP = Math.min(1, Math.max(0, goodP));
    rareP = Math.min(1, Math.max(0, rareP));
    epicP = Math.min(1, Math.max(0, epicP));

    if (epicP > 0) return 'rainbow';
    if (rareP > 0) return 'gold';
    if (goodP > 0) return 'blue';
    return 'white';
  }

  dig(boxId: string, holdSec: number): DigResult {
    if (this.phase !== 'location' || this.location == null) {
      return { ok: false, reason: '当前不能挖', items: [], mishap: false, glow: 'white', staminaCost: 0, quip: '' };
    }
    const locId = this.location;
    const boxes = this.boxes.get(locId) ?? [];
    const box = boxes.find((b) => b.id === boxId);
    if (!box) {
      return { ok: false, reason: '盒子不存在', items: [], mishap: false, glow: 'white', staminaCost: 0, quip: '' };
    }
    if (box.opened) {
      return { ok: false, reason: '盒子已经被掏过了', items: [], mishap: false, glow: 'white', staminaCost: 0, quip: '' };
    }

    const clampHold = Math.min(Math.max(0, holdSec), this.cfg.maxHoldSec);
    const staminaCost = this.cfg.digStamina[box.size] + clampHold * this.cfg.holdStaminaPerSec;
    this.stamina -= staminaCost;
    this.hunger = Math.max(0, this.hunger - this.cfg.hungerPerDig);
    this.minute += this.cfg.minutesPerDig[box.size] + clampHold * this.cfg.minutesPerHoldSecond;
    box.opened = true;

    if (this.hunger <= 0) {
      this.phase = 'lost';
      this.deathCause = this.makeDeathCause();
      return {
        ok: false,
        reason: '饿晕了',
        items: [],
        mishap: false,
        glow: 'white',
        staminaCost,
        quip: '',
        collapsed: false,
      };
    }

    let mishapProb = 0;
    if (clampHold > this.cfg.mishapStartSec) {
      mishapProb = (clampHold - this.cfg.mishapStartSec) * this.cfg.mishapPerSec;
    }
    if (this.fortune?.effect === 'mishapUp' && this.fortuneLocation === locId) {
      mishapProb += this.cfg.fortuneEffects.mishapUp;
    }
    if (this.raining) {
      mishapProb += this.cfg.rainMishapDelta;
    }
    mishapProb = Math.min(1, Math.max(0, mishapProb));
    const isMishap = this.rng.chance(mishapProb);

    const items: ItemInst[] = [];
    let glow: 'white' | 'blue' | 'gold' | 'rainbow' = 'white';
    let quip = '';

    if (isMishap) {
      const mishapDef = this.rng.pick(ITEMS.filter((i) => i.rarity === 'mishap'));
      items.push({ uid: this.nextUid++, itemId: mishapDef.id });
      this.mood = Math.max(0, this.mood + this.cfg.mishapMood);
      quip = this.makeMishapQuip();
    } else {
      const itemCount = box.size === 'big'
        ? this.rng.int(this.cfg.itemsPerBox.big[0], this.cfg.itemsPerBox.big[1])
        : this.cfg.itemsPerBox.small[0];

      const day1Guarantee = this.day === 1 && this.cfg.day1GuaranteeGood && box.size === 'big' && !this.dayHasGoodOrAbove;
      const fortuneGuarantee =
        this.fortune?.effect === 'bigGuaranteeGood' &&
        this.fortuneLocation === locId &&
        box.size === 'big' &&
        !this.fortuneGuaranteeUsed;
      const needGuarantee = day1Guarantee || fortuneGuarantee;
      let guaranteedFulfilled = false;

      for (let i = 0; i < itemCount; i++) {
        let rarity = this.rollRarity(clampHold, box.size, locId);
        if (rarity === 'common' && needGuarantee && !guaranteedFulfilled) {
          rarity = 'good';
        }
        if (rarity === 'good' || rarity === 'rare' || rarity === 'epic') {
          this.dayHasGoodOrAbove = true;
          if (needGuarantee && rarity === 'good') {
            guaranteedFulfilled = true;
          }
        }
        if (rarity === 'rare' || rarity === 'epic') {
          this.mood = Math.min(this.cfg.maxMood, this.mood + this.cfg.rareMood);
        }

        const def = this.pickItemByRarity(rarity, locId);
        items.push({ uid: this.nextUid++, itemId: def.id });
        if (def.edible) {
          this.todayEdibleSeen.push(def.id);
        }
        if (rarity === 'epic') {
          glow = 'rainbow';
        } else if (rarity === 'rare' && glow !== 'rainbow') {
          glow = 'gold';
        } else if (rarity === 'good' && glow === 'white') {
          glow = 'blue';
        }
      }

      if (needGuarantee && guaranteedFulfilled) {
        if (fortuneGuarantee) {
          this.fortuneGuaranteeUsed = true;
        }
      }
      quip = this.makeQuip(items);
    }

    if (this.catLuck > 0) {
      this.catLuck--;
    }

    this.floor.push(...items);

    let event: PendingEvent | undefined;
    let collapsed = false;

    if (!isMishap && this.rng.chance(this.cfg.eventChancePerDig)) {
      event = this.tryTriggerEvent(locId);
      if (event) {
        this.pendingEvent = event;
        this.phase = 'event';
      }
    }

    if (this.stamina <= 0) {
      this.collapse();
      collapsed = true;
      event = undefined;
      this.pendingEvent = null;
    } else {
      this.checkMidnight();
      if ((this.phase as Phase) === 'collapsed') {
        collapsed = true;
        event = undefined;
      }
    }

    return {
      ok: true,
      items,
      mishap: isMishap,
      glow,
      staminaCost,
      quip,
      event,
      collapsed,
    };
  }

  resolveEvent(optionIndex: number, minigameWin: boolean = false): void {
    if (this.phase !== 'event' || !this.pendingEvent) return;
    const ev = this.pendingEvent;
    const locId = this.location;
    this.pendingEvent = null;

    if (!locId) {
      this.phase = 'map';
      return;
    }

    switch (ev.id) {
      case 'patrol': {
        if (optionIndex === 0) {
          this.stamina = Math.max(0, this.stamina - this.cfg.patrolLeaveStamina);
          this.bannedToday.push(locId);
          this.phase = 'map';
        } else {
          if (this.rng.chance(0.5)) {
            this.phase = 'location';
          } else {
            this.mood = Math.max(0, this.mood + this.cfg.patrolMood);
            this.bannedToday.push(locId);
            this.phase = 'map';
          }
        }
        break;
      }
      case 'guard': {
        this.bannedToday.push(locId);
        this.phase = 'map';
        break;
      }
      case 'rival': {
        if (optionIndex === 0) {
          this.mood = Math.max(0, this.mood + this.cfg.rivalYieldMood);
          this.phase = 'location';
        } else {
          if (minigameWin) {
            const boxes = this.boxes.get(locId) ?? [];
            boxes.push({ id: `box_extra_${this.nextUid++}`, size: 'big', opened: false });
            this.phase = 'location';
          } else {
            this.stamina = Math.max(0, this.stamina - this.cfg.rivalLoseStamina);
            this.phase = 'location';
            if (this.stamina <= 0) {
              this.collapse();
            }
          }
        }
        break;
      }
      case 'auntie': {
        this.hunger = Math.min(this.cfg.maxHunger, this.hunger + this.cfg.auntieHunger);
        this.mood = Math.min(this.cfg.maxMood, this.mood + this.cfg.auntieMood);
        this.phase = 'location';
        break;
      }
      case 'delivery': {
        this.hunger = Math.min(this.cfg.maxHunger, this.hunger + this.cfg.deliveryHunger);
        this.phase = 'location';
        break;
      }
      case 'wallet': {
        this.money += this.cfg.walletMoney;
        this.mood = Math.min(this.cfg.maxMood, this.mood + this.cfg.walletMood);
        this.phase = 'location';
        break;
      }
      case 'cat': {
        if (optionIndex === 0) {
          const foodUid = this.findEdibleUid();
          if (foodUid != null) {
            this.eat(foodUid);
            this.catLuck = this.cfg.catLuckDigs;
          }
        }
        this.phase = 'location';
        break;
      }
      case 'rain': {
        this.raining = true;
        this.phase = 'location';
        break;
      }
    }
  }

  canPlace(itemCells: number, x: number, y: number, rotated: boolean): boolean {
    const w = itemCells === 2 ? (rotated ? 1 : 2) : 1;
    const h = itemCells === 2 ? (rotated ? 2 : 1) : 1;
    if (x < 0 || y < 0 || x + w > this.bagW || y + h > this.bagH) return false;
    for (const p of this.bag) {
      if (this.rectsOverlap(x, y, w, h, p.x, p.y, p.w, p.h)) return false;
    }
    return true;
  }

  placeFromFloor(uid: number, x: number, y: number, rotated: boolean = false): boolean {
    const idx = this.floor.findIndex((i) => i.uid === uid);
    if (idx === -1) return false;
    const item = this.floor[idx];
    const def = this.findItemDef(item.itemId);
    if (!def) return false;
    if (!this.canPlace(def.cells, x, y, rotated)) return false;

    const w = def.cells === 2 ? (rotated ? 1 : 2) : 1;
    const h = def.cells === 2 ? (rotated ? 2 : 1) : 1;
    this.floor.splice(idx, 1);
    this.bag.push({ uid: item.uid, itemId: item.itemId, x, y, w, h, scratchValue: item.scratchValue });
    return true;
  }

  moveToFloor(uid: number): boolean {
    const idx = this.bag.findIndex((p) => p.uid === uid);
    if (idx === -1) return false;
    const p = this.bag[idx];
    this.bag.splice(idx, 1);
    this.floor.push({ uid: p.uid, itemId: p.itemId, scratchValue: p.scratchValue });
    return true;
  }

  discard(uid: number): boolean {
    const idx = this.floor.findIndex((i) => i.uid === uid);
    if (idx === -1) return false;
    this.floor.splice(idx, 1);
    return true;
  }

  eat(uid: number): boolean {
    let idx = this.floor.findIndex((i) => i.uid === uid);
    let isBag = false;
    if (idx === -1) {
      idx = this.bag.findIndex((p) => p.uid === uid);
      isBag = true;
    }
    if (idx === -1) return false;

    const item = isBag ? this.bag[idx] : this.floor[idx];
    const def = this.findItemDef(item.itemId);
    if (!def || !def.edible) return false;

    let foodMul = 1;
    if (this.fortune?.effect === 'foodUp') {
      foodMul = 1 + this.cfg.fortuneEffects.foodUp;
    }
    if (def.hunger) {
      this.hunger = Math.min(this.cfg.maxHunger, this.hunger + Math.round(def.hunger * foodMul));
    }
    if (def.stamina) {
      this.stamina = Math.min(this.maxStamina, this.stamina + def.stamina);
    }
    if (def.mood) {
      this.mood = Math.min(this.cfg.maxMood, this.mood + def.mood);
    }

    if (isBag) {
      this.bag.splice(idx, 1);
    } else {
      this.floor.splice(idx, 1);
    }
    return true;
  }

  leaveLocation(): void {
    if (this.location == null) return;
    this.clearFloor('离开时，没装的东西落在了原地');
    this.location = null;
    this.phase = 'map';
  }

  goHome(): boolean {
    if (this.phase === 'won' || this.phase === 'lost' || this.phase === 'collapsed') return false;
    if (this.location == null) {
      if (this.phase === 'home') return true;
      this.phase = 'home';
      this.sellBagAndReport();
      return true;
    }

    const cost = this.homeCost();
    if (this.stamina < cost) return false;
    this.stamina -= cost;
    this.location = null;
    this.phase = 'home';
    this.clearFloor('回家路上，面前的宝贝不见了');
    this.sellBagAndReport();
    return true;
  }

  endDayAway(): void {
    if (!this.goHome()) {
      this.collapse();
    }
  }

  buyFood(): boolean {
    if (this.phase !== 'home') return false;
    if (this.money < this.cfg.foodPrice) return false;
    this.money -= this.cfg.foodPrice;
    this.hunger = Math.min(this.cfg.maxHunger, this.hunger + this.cfg.foodHunger);
    return true;
  }

  expandBag(): boolean {
    if (this.bagLevel >= this.cfg.bagSizes.length - 1) return false;
    const cost = this.cfg.bagSizes[this.bagLevel + 1].cost;
    if (this.money < cost) return false;
    this.money -= cost;
    this.bagLevel++;
    return true;
  }

  upgradeHome(): boolean {
    if (this.homeLevel >= this.cfg.homeLevels.length - 1) return false;
    const cost = this.cfg.homeLevels[this.homeLevel + 1].cost;
    if (this.money < cost) return false;
    this.money -= cost;
    this.homeLevel++;
    return true;
  }

  sleep(): void {
    if (this.phase !== 'home') return;

    if (this.rng.chance(this.cfg.nightWarmthChance)) {
      this.hunger = Math.min(this.cfg.maxHunger, this.hunger + this.cfg.nightWarmthHunger);
      this.mood = Math.min(this.cfg.maxMood, this.mood + this.cfg.warmthMood);
      if (this.lastDayReport) {
        this.lastDayReport.warmth = true;
      }
    }

    this.hunger -= this.dailyHunger;
    this.mood += this.cfg.moodPerDay;
    this.mood = Math.max(0, Math.min(this.cfg.maxMood, this.mood));
    this.stamina = this.maxStamina;

    this.history.push({ day: this.day, money: this.money, hunger: this.hunger, stamina: this.stamina });

    if (this.hunger <= 0) {
      this.phase = 'lost';
      this.deathCause = this.makeDeathCause();
      return;
    }
    if (this.day === this.cfg.totalDays) {
      this.phase = 'won';
      return;
    }

    this.day += 1;
    this.minute = this.cfg.dayStartMinute;
    this.floor = [];
    this.bag = [];
    this.bannedToday = [];
    this.raining = false;
    this.catLuck = 0;
    this.dayHasGoodOrAbove = false;
    this.fortuneGuaranteeUsed = false;
    this.todayEdibleSeen = [];
    this.pendingEvent = null;
    this.lastDayReport = null;
    this.drawFortune();
    this.phase = 'fortune';
  }

  collapse(): void {
    if (this.phase === 'won' || this.phase === 'lost') return;

    const spot = this.chooseSleepSpot();
    const lostItems: string[] = [];
    const loseCount = Math.floor(this.bag.length * this.cfg.collapseLoseRatio);
    for (let i = 0; i < loseCount && this.bag.length > 0; i++) {
      const idx = Math.floor(this.rng.next() * this.bag.length);
      const removed = this.bag.splice(idx, 1)[0];
      const def = this.findItemDef(removed.itemId);
      if (def) lostItems.push(def.name);
    }

    this.clearFloor('倒下后，面前的宝贝消失了');

    this.hunger -= this.dailyHunger;
    this.mood += this.cfg.moodPerDay;
    this.mood = Math.max(0, Math.min(this.cfg.maxMood, this.mood));

    if (this.hunger <= 0) {
      this.phase = 'lost';
      this.deathCause = this.makeDeathCause();
      return;
    }

    let finalSpot = spot;
    let staminaNext = spot.stamina;
    if (spot.cost > 0) {
      if (this.money >= spot.cost) {
        this.money -= spot.cost;
      } else {
        finalSpot = SLEEP_SPOTS[0];
        staminaNext = finalSpot.stamina;
        if (this.rng.chance(this.cfg.benchGuardChance)) {
          staminaNext -= this.cfg.benchGuardStamina;
        }
      }
    } else {
      if (finalSpot.id === 'bench' && this.rng.chance(this.cfg.benchGuardChance)) {
        staminaNext -= this.cfg.benchGuardStamina;
      }
    }
    staminaNext = Math.max(0, staminaNext);
    this.stamina = staminaNext;

    this.lastCollapse = {
      day: this.day,
      spot: finalSpot.id,
      spotName: finalSpot.name,
      lostItems,
      staminaNextDay: staminaNext,
      text: this.makeCollapseText(finalSpot.name, lostItems),
    };
    this.phase = 'collapsed';
    this.pendingEvent = null;
  }

  wakeUp(): void {
    if (this.phase !== 'collapsed') return;
    if (this.day === this.cfg.totalDays) {
      this.phase = 'won';
      return;
    }

    this.history.push({ day: this.day, money: this.money, hunger: this.hunger, stamina: this.stamina });
    this.day += 1;
    this.minute = this.cfg.dayStartMinute;
    this.floor = [];
    this.bag = [];
    this.bannedToday = [];
    this.raining = false;
    this.catLuck = 0;
    this.dayHasGoodOrAbove = false;
    this.fortuneGuaranteeUsed = false;
    this.todayEdibleSeen = [];
    this.pendingEvent = null;
    this.lastCollapse = null;
    this.location = null;
    this.drawFortune();
    this.phase = 'fortune';
  }

  private drawFortuneInternal(): void {
    const previousFortune = this.fortune;
    let goodWeight = this.cfg.fortuneWeights.good;
    let mehWeight = this.cfg.fortuneWeights.meh;
    let badWeight = this.cfg.fortuneWeights.bad;

    if (previousFortune?.effect === 'nextGoodUp') {
      goodWeight += this.cfg.fortuneEffects.nextGoodUp;
    }

    const phaseDef = this.currentPhaseDef();
    if (phaseDef.badFortuneBonus > 0) {
      badWeight += phaseDef.badFortuneBonus;
    }

    const total = goodWeight + mehWeight + badWeight;
    const roll = this.rng.next() * total;
    let tier: FortuneTier;
    if (roll < goodWeight) {
      tier = 'good';
    } else if (roll < goodWeight + mehWeight) {
      tier = 'meh';
    } else {
      tier = 'bad';
    }

    const candidates = FORTUNES.filter((f) => f.tier === tier);
    const fortune = this.rng.pick(candidates);

    this.fortune = fortune;
    if (fortune.needsLocation) {
      this.fortuneLocation = this.rng.pick(LOCATIONS).id;
    } else {
      this.fortuneLocation = null;
    }
  }

  private currentPhaseDef(): PhaseDef {
    for (const phase of this.cfg.phases) {
      if (this.day >= phase.fromDay && this.day <= phase.toDay) {
        return phase;
      }
    }
    return this.cfg.phases[this.cfg.phases.length - 1];
  }

  private findLocation(id: string): LocationDef | undefined {
    return LOCATIONS.find((loc) => loc.id === id);
  }

  private findBox(boxId: string): Box | undefined {
    for (const boxes of this.boxes.values()) {
      for (const box of boxes) {
        if (box.id === boxId) {
          return box;
        }
      }
    }
    return undefined;
  }

  private rollRarity(holdSec: number, boxSize: BoxSize, locId: string): Rarity {
    const clampHold = Math.min(Math.max(0, holdSec), this.cfg.maxHoldSec);
    let goodP = 0;
    let rareP = 0;
    let epicP = 0;

    for (const step of this.cfg.holdCurve) {
      if (step.minSec <= clampHold) {
        goodP = step.good;
        rareP = step.rare;
        epicP = step.epic;
      } else {
        break;
      }
    }

    const loc = this.findLocation(locId);
    if (!loc) return 'common';
    const phaseDef = this.currentPhaseDef();
    let mul = loc.lootRate * phaseDef.lootMul;

    if (this.fortune?.effect === 'lootUp' && this.fortuneLocation === locId) {
      mul *= 1 + this.cfg.fortuneEffects.lootUp;
    }

    if (this.catLuck > 0) {
      mul *= 1 + this.cfg.catLuckMul;
    }

    if (this.mood < this.cfg.moodGoldLow.threshold) {
      mul *= this.cfg.moodGoldLow.mul;
    } else if (this.mood >= this.cfg.moodGoldHigh.threshold) {
      mul *= this.cfg.moodGoldHigh.mul;
    }

    goodP *= mul;
    rareP *= mul;
    epicP *= mul;

    if (boxSize === 'big') {
      goodP += this.cfg.bigBoxBonusGood;
    }

    goodP = Math.min(1, Math.max(0, goodP));
    rareP = Math.min(1, Math.max(0, rareP));
    epicP = Math.min(1, Math.max(0, epicP));

    const roll = this.rng.next();
    if (roll < epicP) {
      return 'epic';
    } else if (roll < epicP + rareP) {
      return 'rare';
    } else if (roll < epicP + rareP + goodP) {
      return 'good';
    } else {
      return 'common';
    }
  }

  private pickItemByRarity(rarity: Rarity, locId: string): ItemDef {
    const candidates = ITEMS.filter((item) => item.rarity === rarity);
    if (candidates.length === 0) return ITEMS[0];

    const weights = candidates.map((item) => {
      return item.locations?.includes(locId) ? 3 : 1;
    });

    let totalWeight = 0;
    for (const weight of weights) {
      totalWeight += weight;
    }

    let roll = this.rng.next() * totalWeight;
    for (let i = 0; i < candidates.length; i++) {
      if (roll < weights[i]) {
        return candidates[i];
      }
      roll -= weights[i];
    }

    return candidates[candidates.length - 1];
  }

  private makeQuip(items: ItemInst[]): string {
    if (items.length === 0) return '';
    const names = items.map((i) => this.findItemDef(i.itemId)?.name ?? '未知');
    const joined = names.join('、');
    return `掏出：${joined}，深圳欢迎你`;
  }

  private makeMishapQuip(): string {
    const quips = [

      '猫主子给了你一巴掌',
      '今天的垃圾桶格外暴躁',
      '你被垃圾教育了',
    ];
    return this.rng.pick(quips);
  }

  private tryTriggerEvent(locId: string): PendingEvent | undefined {
    const available: EventDef[] = [];
    const weights: number[] = [];

    for (const ev of EVENTS) {
      let weight = ev.weight;
      let allowed = true;

      if (ev.id === 'patrol') {
        const loc = this.findLocation(locId);
        if (loc) {
          weight *= loc.danger;
        }
        weight *= 1 + this.currentPhaseDef().patrolBonus;
        if (this.fortune?.effect === 'patrolX2' && this.fortuneLocation === locId) {
          weight *= this.cfg.fortuneEffects.patrolX2;
        }
      }

      if (ev.id === 'rival' && !this.currentPhaseDef().rivalEnabled) {
        allowed = false;
      }

      if (ev.id === 'rain' && this.raining) {
        allowed = false;
      }

      if (allowed && weight > 0) {
        available.push(ev);
        weights.push(weight);
      }
    }

    if (available.length === 0) return undefined;

    let totalWeight = 0;
    for (const weight of weights) {
      totalWeight += weight;
    }

    let roll = this.rng.next() * totalWeight;
    for (let i = 0; i < available.length; i++) {
      if (roll < weights[i]) {
        const ev = available[i];
        return { id: ev.id, name: ev.name, options: [...ev.options] };
      }
      roll -= weights[i];
    }

    return undefined;
  }

  private findEdibleUid(): number | null {
    for (const item of this.floor) {
      const def = this.findItemDef(item.itemId);
      if (def?.edible) return item.uid;
    }
    for (const item of this.bag) {
      const def = this.findItemDef(item.itemId);
      if (def?.edible) return item.uid;
    }
    return null;
  }

  private rectsOverlap(
    x1: number, y1: number, w1: number, h1: number,
    x2: number, y2: number, w2: number, h2: number,
  ): boolean {
    return x1 < x2 + w2 && x1 + w1 > x2 && y1 < y2 + h2 && y1 + h1 > y2;
  }

  private clearFloor(text: string): void {

      this.floor = [];
    }
  }

  private sellBagAndReport(): void {
    let income = 0;
    const soldItems: string[] = [];
    let bestItem: string | undefined;
    let bestPrice = -Infinity;

    for (const item of this.bag) {
      const def = this.findItemDef(item.itemId);
      if (!def) continue;

      let price = def.price;
      if (def.id === 'scratch_card') {
        price = this.revealScratchValue();
        item.scratchValue = price;
      } else if (def.priceMax != null && def.priceMax > def.price) {
        price = def.price;
      }

      income += price;
      soldItems.push(def.name);

      if (price > bestPrice) {
        bestPrice = price;
        bestItem = def.name;
      }
    }

    this.money += income;
    this.totalIncome += income;
    this.bag = [];

    const report: DayReport = {
      day: this.day,
      income,
      soldItems,
      bestItem,
      hungerLeft: this.hunger,
      quip: this.makeDayQuip(soldItems, income),
    };

    if (this.lastDayReport?.warmth) {
      report.warmth = true;
    }

    this.lastDayReport = report;
    this.log.push(`第 ${this.day} 天卖出：${soldItems.join('、') || '无'}，收入 ¥${income.toFixed(2)}`);
  }

  private revealScratchValue(): number {
    const roll = this.rng.next() * 100;
    if (roll < 60) {
      return 0;
    } else if (roll < 90) {
      return this.rng.int(5, 20);
    } else if (roll < 99) {
      return this.rng.int(30, 80);
    } else {
      return 200;
    }
  }

  private makeDayQuip(soldItems: string[], income: number): string {
    if (soldItems.length === 0) {
      return '今天一无所获，深圳的天空依旧灰蒙蒙';
    }
    const best = soldItems[soldItems.length - 1];
    return `今日战绩：${best}，收入 ¥${income.toFixed(2)}，深圳欢迎你`;
  }

  private chooseSleepSpot(): SleepSpotDef {
    let roll = this.rng.next();
    for (const spot of SLEEP_SPOTS) {
      if (roll < spot.prob) {
        return spot;
      }
      roll -= spot.prob;
    }
    return SLEEP_SPOTS[0];
  }

  private makeDeathCause(): string {
    const dayText = `第 ${this.day} 天`;
    if (this.todayEdibleSeen.length > 0) {
      const name = this.findItemDef(this.todayEdibleSeen[this.todayEdibleSeen.length - 1])?.name ?? '可吃的东西';
      return `${dayText}，舍不得吃那块${name}，饿晕了`;
    }
    const causes = [
      `${dayText}，在深圳的街头饿成了纸片人`,
      `${dayText}，以为喝西北风能饱，结果风停了`,
      `${dayText}，终于理解了什么叫“饿得前胸贴后背”`,
      `${dayText}，塑料袋翻了个遍，只找到了孤独`,
    ];
    return this.rng.pick(causes);
  }

  private makeCollapseText(spotName: string, lostItems: string[]): string {
    if (lostItems.length === 0) {
      return `你在${spotName}睡着了，醒来还是身无分文，但至少没丢东西。`;
    }
    const names = lostItems.join('、');
    return `你在${spotName}睡着了，醒来少了 ${lostItems.length} 样东西：${names}。`;
  }

  private findItemDef(id: string): ItemDef | undefined {
    return ITEMS.find((item) => item.id === id);
  }
  private checkMidnight(): void {
    if (this.phase === 'won' || this.phase === 'lost' || this.phase === 'collapsed' || this.phase === 'home') return;
    if (this.location == null) return;
    if (this.minute < this.cfg.midnightMinute) return;

    if (this.stamina >= this.homeCost()) {
      this.goHome();
    } else {
      this.collapse();
    }
  }

}

/** 文件名自然排序（0.png, 1.png, ... 10.png 不会排到 2 前面） */
function compareNames(a: string, b: string): number {
  if (a === b) return 0;
  const pa = a.replace(/\d+/g, '#');
  const pb = b.replace(/\d+/g, '#');
  if (pa !== pb) return pa < pb ? -1 : 1;
  const na = a.match(/\d+/g);
  const nb = b.match(/\d+/g);
  if (na && nb && na.length > 0 && nb.length > 0) {
    const len = Math.min(na.length, nb.length);
    for (let i = 0; i < len; i++) {
      const x = parseInt(na[i], 10);
      const y = parseInt(nb[i], 10);
      if (x !== y) return x - y;
    }
    if (na.length !== nb.length) return na.length - nb.length;
  }
  return a < b ? -1 : a > b ? 1 : 0;
}

    this._index = 0;
    this.applyFrame();
    this.restartSchedule();
  }

  public get frameCount(): number {
    return this._frames.length;
  }

  protected onEnable(): void {
    this.restartSchedule();
  }

  protected onDisable(): void {
    this.unschedule(this.tick);
  }

  private ensureSprite(): Sprite | null {
    if (!this._sprite || !this._sprite.isValid) {
      this._sprite = this.getComponent(Sprite);
    }
    return this._sprite;
  }

  private applyFrame(): void {
    const sp = this.ensureSprite();
    if (!sp) return;
    sp.spriteFrame =
      this._frames.length > 0
        ? this._frames[Math.min(this._index, this._frames.length - 1)]
        : null;
  }

  private restartSchedule(): void {
    this.unschedule(this.tick);
    if (!this.node || !this.node.activeInHierarchy) return;
    if (this._frames.length <= 1) return;
    this.schedule(this.tick, 1 / this._fps);
  }

  private tick(): void {
    if (this._frames.length <= 1) return;
    this._index++;
    if (this._index >= this._frames.length) {
      if (this._loop) {
        this._index = this._loopFrom;
      } else {
        this._index = this._frames.length - 1;
        this.unschedule(this.tick);
        this.applyFrame();
        return;
      }
    }
    this.applyFrame();
  }
}

  /** 预加载 resources/art 下所有 SpriteFrame，并后台加载背景 bundle。 */
  public static async preload(): Promise<void> {
    if (Art._preloaded) return;
    Art._preloaded = true;

    // 1. 加载 resources/art 下所有 SpriteFrame
    try {
      await new Promise<void>((resolve) => {
        try {
          resources.loadDir(ART_ROOT, SpriteFrame, (err: Error | null, assets: SpriteFrame[]) => {
            if (err || !assets) {
              resolve();
              return;
            }
            Art.collectResFrames();
            resolve();
          });
        } catch (e) {
          resolve();
        }
      });
    } catch (e) {
      // ignore
    }

    // 2. 设置纹理过滤与九宫格 insets
    Art.applyFiltersAndInsets();

    // 3. 后台加载背景 bundle
    await Art.loadBackgrounds();

  /** 获取逻辑名对应的序列帧（bg/icon 映射到正确路径）。无素材返回空数组。 */
  public static frames(name: string): SpriteFrame[] {
    if (!name) return [];

    // icon_<itemId>
    if (name.startsWith('icon_')) {
      const itemId = name.slice(5);
      const iconName = ICON_MAP[itemId];
      if (!iconName) {
        Art.recordMissing(name);
        return [];
      }
      const key = `icons/${iconName}`;
      const list = Art._resFramesCache.get(key);
      if (list && list.length > 0) return list;
      if (Art._preloaded) Art.recordMissing(name);
      return [];
    }

    // bg_<locationId>
    if (name.startsWith('bg_')) {
      const alias = ART_ALIAS[name];
      if (!alias || alias.src !== 'bg') {
        Art.recordMissing(name);
        return [];
      }
      const sf = Art._bgFramesCache.get(alias.path);
      if (sf) return [sf];
      if (Art._bgLoaded) Art.recordMissing(name);
      return [];
    }

    // 普通逻辑名
    const alias = ART_ALIAS[name];
    if (alias && alias.src === 'res') {
      const list = Art._resFramesCache.get(alias.path);
      if (list && list.length > 0) return list;
      if (Art._preloaded) Art.recordMissing(name);
      return [];
    }

    // 兼容：直接尝试用原名作为素材目录
    const legacy = Art._resFramesCache.get(name);
    if (legacy && legacy.length > 0) return legacy;
    if (Art._preloaded) Art.recordMissing(name);
    return [];
  }

   */
  public static makeVisual(
    name: string,
    w: number,
    h: number,
    color: Color,
    label?: string,
  ): Node {
    const width = Math.max(1, Math.round(w));
    const height = Math.max(1, Math.round(h));
    const node = new Node(name || 'visual');
    const ut = node.addComponent(UITransform);
    ut.setContentSize(width, height);

    const frames = Art.frames(name);
    if (frames.length > 0) {
      Art.applySpriteToNode(node, name, frames);
      return node;
    }

    // 背景尚未加载完成？
    const alias = ART_ALIAS[name];
    if (alias && alias.src === 'bg' && !Art._bgLoaded) {
      // 先画占位，等背景加载完成后再替换
      Art.drawPlaceholder(node, width, height, color, label);
      Art._bgWaiters.push({ name, node });
      return node;
    }

    // 普通占位
    Art.drawPlaceholder(node, width, height, color, label);
    return node;
  }

  private static applySpriteToNode(
    node: Node,
    name: string,
    frames: SpriteFrame[],
    fps?: number,
    loop?: boolean,
    loopFrom?: number,
  ): void {
    if (!node || !node.isValid || frames.length === 0) return;

    // 隐藏占位元素
    const g = node.getComponent(Graphics);
    if (g) g.enabled = false;
    const label = node.getChildByName('ph_label');
    if (label) label.active = false;

    let sp = node.getComponent(Sprite);
    if (!sp) {
      sp = node.addComponent(Sprite);
    }
    sp.sizeMode = Sprite.SizeMode.CUSTOM;
    const alias = ART_ALIAS[name];
    if (alias?.insets) {
      sp.type = Sprite.Type.SLICED;
    } else {
      sp.type = Sprite.Type.SIMPLE;
    }
    // 注意：这里假设 UITransform 已经存在并设置了尺寸
    // 如果没有，makeVisual 里已设置；play 升级的节点一般也有了

    const effectiveFps = fps || DEFAULT_FPS;
    const effectiveLoop = loop ?? true;
    const effectiveLoopFrom = loopFrom ?? alias?.loopFrom ?? 0;

    let player = node.getComponent(ArtSpritePlayer);
    if (!player) {
      player = node.addComponent(ArtSpritePlayer);
    }
    player.setFrames(frames, effectiveFps, effectiveLoop, effectiveLoopFrom);
  }

  /** 从已加载的 resources/art 资源中归类缓存。 */
  private static collectResFrames(): void {
    Art._resFramesCache.clear();
    const buckets: Map<string, { key: string; sf: SpriteFrame }[]> = new Map();
    let infos: any[] = [];
    try {
      const got = (resources as any).getDirWithPath(ART_ROOT, SpriteFrame) as any[];
      if (got && got.length > 0) infos = got;
    } catch (e) {
      infos = [];
    }

    for (let i = 0; i < infos.length; i++) {
      const info: any = infos[i];
      if (!info) continue;
      const rawPath: string = info.path || info.url || '';
      if (!rawPath) continue;

      const segs = rawPath.split('/');
      let parts = segs;
      const tail = parts[parts.length - 1];
      if (tail === 'spriteFrame' || tail === 'texture') {
        parts = parts.slice(0, parts.length - 1);
      }

      if (parts.length < 3) continue; // 至少 art/xxx/file

      const artIdx = parts.indexOf(ART_ROOT);
      if (artIdx === -1) continue; // 不在 art 目录下
      const relParts = parts.slice(artIdx + 1); // 相对 art/ 的路径段
      if (relParts.length < 2) continue; // 至少有一个目录和文件

      let name: string;
      let file: string;
      if (relParts.length === 2) {
        // 目录/文件 或 子目录/文件（如 icons/plastic_bottle）
        const dirName = relParts[0];
        file = relParts[1];
        if (dirName === 'icons') {
          // 图标单独作为 key：icons/<filename>
          name = `icons/${file}`;
        } else {
          name = dirName;
        }
      } else if (relParts.length > 2) {
        // 多级目录，暂时只取最后一级目录名作为 key
        name = relParts[relParts.length - 2];
        file = relParts[relParts.length - 1];
      } else {
        continue;
      }

      let sf: SpriteFrame | null = null;
      try {
        sf = resources.get(rawPath, SpriteFrame) as SpriteFrame | null;
      } catch (e) {
        sf = null;
      }
      if (!sf) {
        // 尝试直接使用 info.asset
        sf = info.asset as SpriteFrame | null;
      }
      if (!sf) continue;

      let arr = buckets.get(name);
      if (!arr) {
        arr = [];
        buckets.set(name, arr);
      }
      arr.push({ key: file, sf });
    }

    buckets.forEach((arr: { key: string; sf: SpriteFrame }[], name: string) => {
      arr.sort((a, b) => compareNames(a.key, b.key));
      const frames: SpriteFrame[] = [];
      for (let i = 0; i < arr.length; i++) frames.push(arr[i].sf);
      if (frames.length > 0) Art._resFramesCache.set(name, frames);
    });
  }

  /** 设置纹理过滤与九宫格 insets。 */
  private static applyFiltersAndInsets(): void {
    // 设置所有 res SpriteFrame 的 filter
    Art._resFramesCache.forEach((frames: SpriteFrame[]) => {
      frames.forEach((sf: SpriteFrame) => {
        if (sf.texture) {
          (sf.texture as Texture2D).setFilters(
            Texture2D.Filter.NEAREST,
            Texture2D.Filter.NEAREST,
          );
        }
      });
    });

    // 根据 ART_ALIAS 设置 insets
    for (const [name, alias] of Object.entries(ART_ALIAS)) {
      if (alias.src !== 'res' || !alias.insets) continue;
      const frames = Art._resFramesCache.get(alias.path);
      if (!frames || frames.length === 0) continue;
      const [l, r, t, b] = alias.insets;
      frames.forEach((sf: SpriteFrame) => {
        sf.insetLeft = l;
        sf.insetRight = r;
        sf.insetTop = t;
        sf.insetBottom = b;
      });
    }
  }

  /** 加载背景 bundle 并缓存所有背景。 */
  private static async loadBackgrounds(): Promise<void> {
    return new Promise<void>((resolve) => {
      assetManager.loadBundle('bg', (err: Error | null, bundle: AssetManager.Bundle | null) => {
        if (err || !bundle) {
          Art._bgLoaded = true;
          Art._onBgLoaded();
          resolve();
          return;
        }

        const bgAliases = Object.entries(ART_ALIAS).filter(
          ([, alias]) => alias.src === 'bg',
        );

        let remaining = bgAliases.length;
        if (remaining === 0) {
          Art._bgLoaded = true;
          Art._onBgLoaded();
          resolve();
          return;
        }

        const tryLoad = (alias: { path: string }) => {
          const path = alias.path;
          const loadByPath = (url: string) => {
            bundle.load(url, SpriteFrame, (err2: Error | null, sf: SpriteFrame | null) => {
              if (err2 || !sf) {
                // 尝试另一种方式
                const altUrl = url === `${path}/spriteFrame` ? path : `${path}/spriteFrame`;
                if (url !== altUrl) {
                  tryLoad({ path });
                } else {
                  finishOne(null);
                }
                return;
              }
              finishOne(sf);
            });
          };
          loadByPath(`${path}/spriteFrame`);
        };

        const finishOne = (sf: SpriteFrame | null) => {
          if (sf) {
            if (sf.texture) {
              (sf.texture as Texture2D).setFilters(
                Texture2D.Filter.NEAREST,
                Texture2D.Filter.NEAREST,
              );
            }
            Art._bgFramesCache.set(alias.path, sf);
          }
          remaining--;
          if (remaining <= 0) {
            Art._bgLoaded = true;
            Art._onBgLoaded();
            resolve();
          }
        };

        for (const [, alias] of bgAliases) {
          tryLoad(alias);
        }
      });
    });
  }

  /** 背景加载完成后的处理：替换等待中的占位节点。 */
  private static _onBgLoaded(): void {
    const waiters = Art._bgWaiters.slice();
    Art._bgWaiters = [];
    for (const { name, node } of waiters) {
      const frames = Art.frames(name);
      if (frames.length > 0 && node && node.isValid) {
        Art.applySpriteToNode(node, name, frames);
      }
    }
  }
}```

/**
 * Fx —— 纯代码缓动 / 特效工具。
 * 所有节点都是 new Node() 现造，父节点由调用方（GameRoot / 各 View）传入。
 * 不依赖 prefab、不依赖场景里预先摆放的任何东西。
 */
export class Fx {
  // ---------------------------------------------------------------- 抖动
  /** 小幅左右/上下抖动 */
  static shake(node: Node, strength: number = 10, duration: number = 0.3): void {
    if (!node || !node.isValid) return;
    Tween.stopAllByTarget(node);
    const base = node.position.clone();
    const steps = 6;
    const seg = Math.max(0.016, duration / (steps + 1));
    let tw = tween(node);
    for (let i = 0; i < steps; i++) {
      const k = strength * (1 - i / steps);
      tw = tw.to(seg, {
        position: new Vec3(
          base.x + (Math.random() * 2 - 1) * k,
          base.y + (Math.random() * 2 - 1) * k,
          base.z,
        ),
      });
    }
    tw.to(seg, { position: base }).start();
  }

  /** 整屏震一下（一般给 Canvas 根节点） */
  static screenShake(root: Node): void {
    if (!root || !root.isValid) return;
    Fx.shake(root, 18, 0.4);
  }

  /** 鼓一下再回来 */
  static pulse(node: Node, scale: number = 1.12, duration: number = 0.12): void {
    if (!node || !node.isValid) return;
    Tween.stopAllByTarget(node);
    const base = node.scale.clone();
    const half = Math.max(0.03, duration / 2);
    tween(node)
      .to(half, { scale: new Vec3(base.x * scale, base.y * scale, base.z) })
      .to(half, { scale: base })
      .start();
  }

  // ---------------------------------------------------------------- 抛物线
  /** 从 from 抛物线飞到 to（x 线性、y 叠加一个正弦拱高） */
  static popArc(node: Node, from: Vec3, to: Vec3, duration: number = 0.45): Promise<void> {
    return new Promise<void>((resolve) => {
      if (!node || !node.isValid) {
        resolve();
        return;
      }
      const fz = from.z || 0;
      const tz = to.z || 0;
      node.setPosition(from.x, from.y, fz);
      const arc = Math.max(60, Math.min(240, Math.abs(to.x - from.x) * 0.35 + 60));
      const state = { t: 0 };
      tween(state)
        .to(
          Math.max(0.05, duration),
          { t: 1 },
          {
            onUpdate: () => {
              if (!node || !node.isValid) return;
              const t = state.t;
              const x = from.x + (to.x - from.x) * t;
              const y = from.y + (to.y - from.y) * t + Math.sin(Math.PI * t) * arc;
              node.setPosition(x, y, fz + (tz - fz) * t);
            },
          },
        )
        .call(() => {
          if (node && node.isValid) node.setPosition(to.x, to.y, tz);
          resolve();
        })
        .start();
    });
  }

  // ---------------------------------------------------------------- 时间
  /** 冻结（等待）指定秒数 */
  static freeze(seconds: number): Promise<void> {
    return new Promise<void>((resolve) => {
      const state = { t: 0 };
      tween(state)
        .delay(Math.max(0, seconds))
        .call(() => resolve())
        .start();
    });
  }

  // ---------------------------------------------------------------- 闪屏 / 光晕
  /** 全屏闪一下颜色（默认白） */
  static flash(root: Node, color: Color = new Color(255, 255, 255, 150), duration: number = 0.35): void {
    if (!root || !root.isValid) return;
    const size = view.getVisibleSize();
    const w = size.width;
    const h = size.height;
    const n = new Node('fx_flash');
    n.parent = root;
    const ut = n.addComponent(UITransform);
    ut.setContentSize(w, h);
    const g = n.addComponent(Graphics);
    g.fillColor = color ? color.clone() : new Color(255, 255, 255, 150);
    g.rect(-w / 2, -h / 2, w, h);
    g.fill();

    const row = Math.floor(col / cols);
    const colIdx = col % cols;
    const baseX = (colIdx - (Math.min(cols, this.game.boxesAt(locId).length) - 1) / 2) * spacingX;
    const baseY = 0 - row * spacingY;
    boxNode.setPosition(
      baseX + (Math.random() - 0.5) * strength,
      baseY + (Math.random() - 0.5) * strength,
      0,
    );

    // 碎纸片
    const now = Date.now();
    if (now - this.lastDebrisTime > 120) {
      this.lastDebrisTime = now;
      this.spawnDebris(boxNode.getWorldPosition());
    }

    // 光晕
    const glow = this.game.glowFor(sec, boxId);
    this.updateGlow(glow, boxNode.getWorldPosition());

    // 进度条和文字
    const ratio = sec / CONFIG.maxHoldSec;
    if (this.progressFill) Fx.barTo(this.progressFill, ratio);
    const staminaCost = CONFIG.digStamina[box.size] + sec * CONFIG.holdStaminaPerSec;
    const remaining = this.game.stamina;
    if (this.holdLabel) {
      this.holdLabel.string = `剩余体力 ${remaining} / 本次将扣 ${staminaCost.toFixed?.(1)}`;
    }

    // edge glow
    if (glow === 'gold' || glow === 'rainbow') {
      if (!this.edgeGlowOn) {
        Fx.edgeGlow(this.root, glow === 'gold' ? new Color(255, 220, 100, 220) : new Color(255, 100, 255, 220), true);
        this.edgeGlowOn = true;
      }
    } else {
      if (this.edgeGlowOn) {
        Fx.edgeGlow(this.root, new Color(255, 255, 255, 255), false);
        this.edgeGlowOn = false;
      }
    }
  }

  private spawnDebris(worldPos: Vec3): void {
    if (!this.viewNode) return;
    const pieces = 3;
    for (let i = 0; i < pieces; i++) {
      const n = new Node('debris');
      n.parent = this.viewNode;
      const ut = n.addComponent(UITransform);
      const w = 8 + Math.random() * 10;
      ut.setContentSize(w, w);
      const g = n.addComponent(Graphics);
      g.fillColor = new Color(180, 160, 120, 255);
      g.rect(-w / 2, -w / 2, w, w);
      g.fill();

      // 将世界坐标转换为 viewNode 局部坐标
      const local = this.viewNode.getComponent(UITransform)!.convertToNodeSpaceAR(worldPos);
      n.setPosition(local.x, local.y, 0);

      const angle = Math.random() * Math.PI * 2;
      const dist = 60 + Math.random() * 80;
      const target = new Vec3(local.x + Math.cos(angle) * dist, local.y + Math.sin(angle) * dist, 0);
      Fx.popArc(n, n.position.clone(), target, 0.5).then(() => {
        if (n.isValid) n.destroy();
      });
    }
  }

  private updateGlow(glow: 'white' | 'blue' | 'gold' | 'rainbow', worldPos: Vec3): void {
    if (!this.viewNode) return;
    if (this.glowNode && this.glowNode.isValid) {
      this.glowNode.destroy();
      this.glowNode = null;
    }
    const colors: Record<string, Color> = {
      white: new Color(255, 255, 255, 180),
      blue: new Color(100, 200, 255, 200),
      gold: new Color(255, 220, 80, 220),
      rainbow: new Color(255, 120, 255, 220),
    };
    const color = colors[glow] ?? colors.white;
    this.glowNode = Art.makeVisual(`glow_${glow}`, 70, 70, color);
    this.glowNode.parent = this.viewNode;
    const local = this.viewNode.getComponent(UITransform)!.convertToNodeSpaceAR(worldPos);
    this.glowNode.setPosition(local.x, local.y + 50, 0);
  }

  private clearHoldEffects(): void {
    if (this.glowNode && this.glowNode.isValid) {
      this.glowNode.destroy();
      this.glowNode = null;
    }
    if (this.edgeGlowOn) {
      Fx.edgeGlow(this.root, new Color(255, 255, 255, 255), false);
      this.edgeGlowOn = false;
    }
  }

  // ------------------------------------------------------------------
  // 挖盒子动画与结果处理
  // ------------------------------------------------------------------
  private async performDig(boxId: string, holdSec: number): Promise<void> {
    const boxNode = this.boxNodes.get(boxId);
    const box = this.game.boxesAt(this.game.location!).find((b) => b.id === boxId);
    if (!boxNode || !box) return;

    // 盒子开启动画
    if (box.size === 'small') {
      Fx.pulse(boxNode, 1.3, 0.15);
      await Fx.freeze(0.2);
      await Fx.fadeOut(boxNode, 0.2);
    } else {
      Fx.pulse(boxNode, 1.5, 0.25);
      await Fx.freeze(0.3);
      // 模拟盖子弹开
      const cap = new Node('cap');
      cap.parent = boxNode;
      const capUt = cap.addComponent(UITransform);
      capUt.setContentSize(20, 20);
      const capG = cap.addComponent(Graphics);
      capG.fillColor = new Color(100, 70, 40, 255);
      capG.circle(0, 0, 10);
      capG.fill();
      cap.setPosition(0, 0, 0);
      Fx.popArc(cap, new Vec3(0, 0, 0), new Vec3(0, 120, 0), 0.4).then(() => cap.destroy());
      await Fx.freeze(0.35);
      await Fx.fadeOut(boxNode, 0.3);
    }
    boxNode.active = false;

    const result = this.game.dig(boxId, holdSec);
    if (!result.ok) {
      if (result.reason) {
        Fx.floatText(this.viewNode!, result.reason, new Vec3(0, 200, 0), new Color(255, 80, 80, 255));
      }
      this.onChanged();
      return;
    }

    // 事故
    if (result.mishap) {
      this.spawnMishapEffect();
      Fx.screenShake(this.root);
    }

    // 物品飞入待选区
    const boxWorld = this.getBoxWorldPos(boxId);
    for (const item of result.items) {
      await this.spawnItemToFloor(item, boxWorld);
    }
    this.layoutFloor();

    // 稀有/史诗特效
    if (result.glow === 'gold' || result.glow === 'rainbow') {
      await Fx.freeze(0.3);
      Fx.flash(this.root, result.glow === 'gold' ? new Color(255, 220, 150, 180) : new Color(255, 150, 255, 180));
      Fx.screenShake(this.root);
      if (result.quip) {
        Fx.floatText(this.viewNode!, result.quip, new Vec3(0, -40, 0), new Color(255, 255, 180, 255));
      }
    } else if (result.quip) {
      Fx.floatText(this.viewNode!, result.quip, new Vec3(0, -40, 0));
    }

    this.onChanged();

    if (result.event) {
      this.onEvent(result.event);
    } else if (result.collapsed) {
      this.onChanged(); // GameRoot 会切换到 collapsed
    }
  }

  private getBoxWorldPos(boxId: string): Vec3 {
    const node = this.boxNodes.get(boxId);
    if (!node) return new Vec3(0, 0, 0);
    return node.getWorldPosition();
  }

  private spawnMishapEffect(): void {
    if (!this.viewNode) return;
    const isCat = Math.random() > 0.5;
    const node = Art.makeVisual(isCat ? 'npc_cat' : 'ev_rain', 80, 80, isCat ? new Color(120, 120, 120, 220) : new Color(100, 200, 100, 200), isCat ? '喵' : '绿烟');
    node.parent = this.viewNode;
    node.setPosition(0, 80, 0);
    Fx.shake(node, 12, 0.5);
    tween(node).delay(1.2).call(() => node.destroy()).start();
  }

  private async spawnItemToFloor(item: ItemInst, fromWorld: Vec3): Promise<void> {
    if (!this.floorContainer || !this.viewNode) return;
    const def = ITEMS.find((i) => i.id === item.itemId);
    const color = this.rarityColor(def?.rarity ?? 'common');
    const node = this.createItemNode(item, def, color);
    const floorLocalFrom = this.floorContainer.getComponent(UITransform)!.convertToNodeSpaceAR(fromWorld);
    const target = this.getFloorSlotPosition(this.game.floor.indexOf(item));
    node.parent = this.floorContainer;
    node.setPosition(floorLocalFrom.x, floorLocalFrom.y, 0);
    this.floorNodes.set(item.uid, node);
    await Fx.popArc(node, floorLocalFrom, target, 0.5);
  }

  private createItemNode(item: ItemInst, def?: ItemDef, color?: Color): Node {
    const fallbackColor = color ?? this.rarityColor(def?.rarity ?? 'common');
    return Art.makeVisual(`icon_${item.itemId}`, 48, 48, fallbackColor, def?.name ?? item.itemId);
  }

  private rarityColor(rarity: Rarity): Color {
    switch (rarity) {
      case 'common': return new Color(170, 170, 170, 255);
      case 'good': return new Color(100, 200, 120, 255);
      case 'rare': return new Color(80, 160, 255, 255);
      case 'epic': return new Color(200, 100, 255, 255);
      case 'mishap': return new Color(220, 100, 100, 255);
      default: return new Color(200, 200, 200, 255);
    }
  }

  private getFloorSlotPosition(index: number): Vec3 {
    const spacing = 70;
    const totalW = (this.game.floor.length - 1) * spacing;
    return new Vec3(-totalW / 2 + index * spacing, 0, 0);
  }

  private layoutFloor(): void {
    if (!this.floorContainer) return;
    const items = this.game.floor;
    for (let i = 0; i < items.length; i++) {
      const uid = items[i].uid;
      const node = this.floorNodes.get(uid);
      if (node) {
        const target = this.getFloorSlotPosition(i);
        tween(node).to(0.2, { position: target }).start();
      }
    }
  }

  // ------------------------------------------------------------------
  // 渲染待选区与背包
  // ------------------------------------------------------------------
  private renderFloor(): void {
    if (!this.floorContainer) return;
    this.floorContainer.children.forEach((child) => child.destroy());
    this.floorNodes.clear();
    this.game.floor.forEach((item, idx) => {
      const def = ITEMS.find((i) => i.id === item.itemId);
      const node = this.createItemNode(item, def);
      node.parent = this.floorContainer;
      const target = this.getFloorSlotPosition(idx);
      node.setPosition(target.x, target.y, 0);
      this.floorNodes.set(item.uid, node);
      this.attachFloorTouch(node, item.uid);
    });
  }

  private renderBag(): void {
    if (!this.bagContainer) return;
    this.bagContainer.children.forEach((child) => child.destroy());
    this.bagNodes.clear();
    this.drawBagGrid();
    this.game.bag.forEach((placed) => {
      const def = ITEMS.find((i) => i.id === placed.itemId);
      const node = this.createItemNode(placed, def);
      node.parent = this.bagContainer;
      const pos = this.placedToLocal(placed.x, placed.y, placed.w, placed.h);
      node.setPosition(pos.x, pos.y, 0);
      this.bagNodes.set(placed.uid, node);
      this.attachBagTouch(node, placed.uid);
    });
  }

  private drawBagGrid(): void {
    if (!this.bagContainer) return;
    const g = this.bagContainer.addComponent(Graphics);
    g.clear();
    const w = this.game.bagW;
    const h = this.game.bagH;
    const totalW = w * CELL;
    const totalH = h * CELL;
    const startX = -totalW / 2;
    const startY = totalH / 2;
    g.strokeColor = new Color(255, 255, 255, 80);
    g.lineWidth = 1;
    for (let i = 0; i <= w; i++) {
      const x = startX + i * CELL;
      g.moveTo(x, startY);
      g.lineTo(x, startY - totalH);
    }
    for (let j = 0; j <= h; j++) {
      const y = startY - j * CELL;
      g.moveTo(startX, y);
      g.lineTo(startX + totalW, y);
    }
    g.stroke();
  }

  private placedToLocal(x: number, y: number, w: number, h: number): Vec3 {
    const totalW = this.game.bagW * CELL;
    const totalH = this.game.bagH * CELL;
    const localX = -totalW / 2 + x * CELL + (w * CELL) / 2;
    const localY = totalH / 2 - y * CELL - (h * CELL) / 2;
    return new Vec3(localX, localY, 0);
  }

  private attachFloorTouch(node: Node, uid: number): void {
    node.on(Node.EventType.TOUCH_START, (event: EventTouch) => this.onFloorTouchStart(event, uid), this);
    node.on(Node.EventType.TOUCH_MOVE, (event: EventTouch) => this.onFloorTouchMove(event, uid), this);
    node.on(Node.EventType.TOUCH_END, (event: EventTouch) => this.onFloorTouchEnd(event, uid), this);
    node.on(Node.EventType.TOUCH_CANCEL, (event: EventTouch) => this.onFloorTouchCancel(event, uid), this);
  }

  private attachBagTouch(node: Node, uid: number): void {
    node.on(Node.EventType.TOUCH_START, (event: EventTouch) => this.onBagTouchStart(event, uid), this);
    node.on(Node.EventType.TOUCH_END, (event: EventTouch) => this.onBagTouchEnd(event, uid), this);
    node.on(Node.EventType.TOUCH_CANCEL, (event: EventTouch) => this.onBagTouchCancel(event, uid), this);
  }

  // ------------------------------------------------------------------
  // 待选区拖拽
  // ------------------------------------------------------------------
  private onFloorTouchStart(event: EventTouch, uid: number): void {
    if (this.draggingUid !== null || this.holdBoxId) return;
    const node = this.floorNodes.get(uid);
    if (!node || !this.floorContainer) return;

    this.potentialDrag = true;
    this.isDragging = false;
    this.touchStartPos = this.getTouchLocal(event, this.floorContainer);
    this.touchStartTime = Date.now();
    this.draggingUid = uid;
    this.draggingNode = node;
    this.dragOffset = new Vec3(node.position.x - this.touchStartPos.x, node.position.y - this.touchStartPos.y, 0);
  }

  private onFloorTouchMove(event: EventTouch, uid: number): void {
    if (this.draggingUid !== uid || !this.potentialDrag || !this.floorContainer || !this.draggingNode) return;
    const touchLocal = this.getTouchLocal(event, this.floorContainer);
    const dist = Vec3.distance(this.touchStartPos, touchLocal);
    if (dist > 20) {
      this.isDragging = true;
      this.potentialDrag = false;
      // 将节点提升到 viewNode 以便拖拽不受容器裁剪影响
      if (this.viewNode) {
        const world = this.draggingNode.getWorldPosition();
        const local = this.viewNode.getComponent(UITransform)!.convertToNodeSpaceAR(world);
        this.draggingNode.setParent(this.viewNode);
        this.draggingNode.setPosition(local.x, local.y, 0);
      }
    }
    if (this.isDragging && this.draggingNode && this.viewNode) {
      const touchView = this.getTouchLocal(event, this.viewNode);
      this.draggingNode.setPosition(touchView.x + this.dragOffset.x, touchView.y + this.dragOffset.y, 0);
    }
  }

  private onFloorTouchEnd(event: EventTouch, uid: number): void {
    if (this.draggingUid !== uid) return;
    const wasDragging = this.isDragging;
    const node = this.draggingNode;
    this.draggingUid = null;
    this.draggingNode = null;
    this.potentialDrag = false;
    this.isDragging = false;

    if (!wasDragging) {
      // 单击打开菜单
      this.showItemMenu(uid, false);
      return;
    }

    // 尝试放置到背包
    if (!node || !this.viewNode || !this.bagContainer) return;
    const touchView = this.getTouchLocal(event, this.viewNode);
    const dropResult = this.tryPlaceToBag(uid, touchView);
    if (dropResult.success) {
      Fx.floatText(this.viewNode, '咔', dropResult.pos, new Color(255, 255, 180, 255));
      this.renderFloor();
      this.renderBag();
      this.onChanged();
    } else {
      // 失败：格子闪红，物品回原位
      this.flashBagCell(dropResult.cell);
      const floorLocal = dropResult.originalFloorPos;
      if (this.floorContainer) {
        node.setParent(this.floorContainer);
        tween(node).to(0.25, { position: floorLocal }).start();
      }
      this.renderFloor(); // 重建以保证位置正确
    }
  }

  private onFloorTouchCancel(event: EventTouch, uid: number): void {
    if (this.draggingUid !== uid) return;
    this.cancelDrag();
  }

  private cancelDrag(): void {
    if (this.draggingNode && this.floorContainer) {
      this.draggingNode.setParent(this.floorContainer);
      // 回到原始 slot
      const uid = this.draggingUid;
      if (uid != null) {
        const idx = this.game.floor.findIndex((i) => i.uid === uid);
        if (idx >= 0) {
          const target = this.getFloorSlotPosition(idx);
          tween(this.draggingNode).to(0.2, { position: target }).start();
        }
      }
    }
    this.draggingUid = null;
    this.draggingNode = null;
    this.potentialDrag = false;
    this.isDragging = false;
  }

  private tryPlaceToBag(uid: number, touchView: Vec3): {
    success: boolean;
    cell: { x: number; y: number } | null;
    pos: Vec3;
    originalFloorPos: Vec3;
  } {
    const item = this.game.floor.find((i) => i.uid === uid);
    if (!item) return { success: false, cell: null, pos: new Vec3(), originalFloorPos: new Vec3() };
    const def = ITEMS.find((i) => i.id === item.itemId);
    if (!def) return { success: false, cell: null, pos: new Vec3(), originalFloorPos: new Vec3() };

    const rotated = this.floorRotated.get(uid) ?? false;
    const cell = this.cellFromViewTouch(touchView);
    if (!cell) return { success: false, cell: null, pos: new Vec3(), originalFloorPos: new Vec3() };

    const success = this.game.placeFromFloor(uid, cell.x, cell.y, rotated);
    const placed = this.game.bag.find((p) => p.uid === uid);
    const pos = placed ? this.placedToLocal(placed.x, placed.y, placed.w, placed.h) : new Vec3();
    const originalIdx = this.game.floor.findIndex((i) => i.uid === uid); // may be -1
    const originalFloorPos = originalIdx >= 0 ? this.getFloorSlotPosition(originalIdx) : new Vec3();
    return { success, cell, pos, originalFloorPos };
  }

  private cellFromViewTouch(touchView: Vec3): { x: number; y: number } | null {
    if (!this.bagContainer) return null;
    // bagContainer 位置 (0, BAG_Y)，touchView 是 viewNode 局部坐标
    const bagLocalX = touchView.x - this.bagContainer.position.x;
    const bagLocalY = touchView.y - this.bagContainer.position.y;
    const totalW = this.game.bagW * CELL;
    const totalH = this.game.bagH * CELL;
    const startX = -totalW / 2;
    const startY = totalH / 2;
    const relX = bagLocalX - startX;
    const relY = startY - bagLocalY;
    if (relX < 0 || relY < 0 || relX >= totalW || relY >= totalH) return null;
    const x = Math.floor(relX / CELL);
    const y = Math.floor(relY / CELL);
    if (x < 0 || x >= this.game.bagW || y < 0 || y >= this.game.bagH) return null;
    return { x, y };
  }

  private flashBagCell(cell: { x: number; y: number } | null): void {
    if (!cell || !this.bagContainer) return;
    const w = CELL;
    const h = CELL;
    const totalW = this.game.bagW * CELL;
    const totalH = this.game.bagH * CELL;
    const startX = -totalW / 2;
    const startY = totalH / 2;
    const x = startX + cell.x * CELL;
    const y = startY - cell.y * CELL - CELL;

    const flash = new Node('cell_flash');
    flash.parent = this.bagContainer;
    const fut = flash.addComponent(UITransform);
    fut.setContentSize(w, h);
    const fg = flash.addComponent(Graphics);
    fg.fillColor = new Color(255, 80, 80, 160);
    fg.rect(-w / 2, -h / 2, w, h);
    fg.fill();
    flash.setPosition(x + w / 2, y + h / 2, 0);
    tween(flash).to(0.2, { scale: new Vec3(1.2, 1.2, 1) }).to(0.2, { scale: new Vec3(1, 1, 1) }).call(() => flash.destroy()).start();
  }

  // ------------------------------------------------------------------
  // 背包物品点击
  // ------------------------------------------------------------------
  private onBagTouchStart(event: EventTouch, uid: number): void {
    // 留空，避免事件穿透
  }

  private onBagTouchEnd(event: EventTouch, uid: number): void {
    if (this.draggingUid !== null) return;
    this.showItemMenu(uid, true);
  }

  private onBagTouchCancel(event: EventTouch, uid: number): void {
    // 留空
  }

  // ------------------------------------------------------------------
  // 物品菜单
  // ------------------------------------------------------------------
  private menuNode: Node | null = null;

  private showItemMenu(uid: number, isBag: boolean): void {
    this.closeMenu();
    if (!this.viewNode) return;

    let def: ItemDef | undefined;
    if (isBag) {
      const placed = this.game.bag.find((p) => p.uid === uid);
      def = placed ? ITEMS.find((i) => i.id === placed.itemId) : undefined;
    } else {
      const item = this.game.floor.find((i) => i.uid === uid);
      def = item ? ITEMS.find((i) => i.id === item.itemId) : undefined;
    }
    if (!def) return;

    const menu = new Node('item_menu');
    menu.parent = this.viewNode;
    const mut = menu.addComponent(UITransform);
    mut.setContentSize(260, 220);
    menu.setPosition(0, 200, 0);
    const dim = new Node('dim');
    dim.parent = menu;
    const dut = dim.addComponent(UITransform);
    dut.setContentSize(720, 1280);
    const dg = dim.addComponent(Graphics);
    dg.fillColor = new Color(0, 0, 0, 120);
    dg.rect(-360, -640, 720, 1280);
    dg.fill();
    dim.on(Node.EventType.TOUCH_START, () => this.closeMenu(), this);

    const panel = new Node('panel');
    panel.parent = menu;
    const put = panel.addComponent(UITransform);
    put.setContentSize(260, 220);
    const pg = panel.addComponent(Graphics);
    pg.fillColor = new Color(40, 40, 40, 235);
    pg.roundRect(-130, -110, 260, 220, 12);
    pg.fill();
    pg.strokeColor = new Color(200, 200, 200, 255);
    pg.lineWidth = 2;
    pg.stroke();

    const title = new Node('title');
    const tut = title.addComponent(UITransform);
    tut.setContentSize(240, 40);
    const tl = title.addComponent(Label);
    tl.string = def.name;
    tl.fontSize = 26;
    tl.lineHeight = 30;
    tl.color = new Color(255, 255, 255, 255);
    tl.horizontalAlign = Label.HorizontalAlign.CENTER;
    tl.verticalAlign = Label.VerticalAlign.CENTER;
    title.parent = panel;
    title.setPosition(0, 80, 0);

    const yStart = 20;
    const yStep = -50;
    let y = yStart;

    if (def.edible) {
      this.addMenuButton(panel, '吃掉', new Vec3(0, y, 0), () => {
        this.game.eat(uid);
        this.closeMenu();
        this.renderFloor();
        this.renderBag();
        this.onChanged();
      });
      y += yStep;
    }

    if (isBag) {
      this.addMenuButton(panel, '拿出', new Vec3(0, y, 0), () => {
        this.game.moveToFloor(uid);
        this.closeMenu();
        this.renderFloor();
        this.renderBag();
        this.onChanged();
      });
      y += yStep;
    } else {
      this.addMenuButton(panel, '扔掉', new Vec3(0, y, 0), () => {
        this.game.discard(uid);
        this.closeMenu();
        this.renderFloor();
        this.onChanged();
      });
      y += yStep;

      if (def.cells === 2) {
        const item = this.game.floor.find((i) => i.uid === uid);
        if (item) {
          const rotated = this.floorRotated.get(uid) ?? false;
          this.addMenuButton(panel, rotated ? '取消旋转' : '旋转', new Vec3(0, y, 0), () => {
            this.floorRotated.set(uid, !rotated);
            this.closeMenu();
            this.renderFloor();
          });
        }
      }
    }

    this.menuNode = menu;
  }

  private addMenuButton(parent: Node, text: string, pos: Vec3, onClick: () => void): void {
